from __future__ import annotations

import argparse
import json
from pathlib import Path
from typing import Any

from fashion_mm.models.attributes import FashionAttributePredictor
from fashion_mm.models.instance_segmentation import FashionInstanceSegmentationPredictor
from fashion_mm.models.instance_segmentation import draw_segmentation_result
from fashion_mm.models.local_region.visualization import draw_local_region_result
from fashion_mm.pipelines import FashionVisualPipeline
from fashion_mm.utils.config import load_config
from fashion_mm.utils.image_io import save_mask


RESULT_SCHEMA_NAME = "fashion_mm.fashion_visual_pipeline_result"
RESULT_SCHEMA_VERSION = "1.0.0"
PACKAGE_FILENAMES = {
    "result_json": "result.json",
    "segmentation_visualization": "segmentation_visualization.jpg",
    "local_region_mask": "local_region_mask.png",
    "local_region_visualization": "local_region_visualization.jpg",
}


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(
        description="Run the 3.1.1 -> 3.1.2 -> 3.1.3 fashion visual pipeline."
    )
    parser.add_argument("image", help="Path to an RGB fashion image.")
    parser.add_argument("query", help="Natural-language target-region query.")
    parser.add_argument(
        "--segmentation-config",
        default="configs/model/instance_segmentation_deepfashion2.yaml",
    )
    parser.add_argument("--segmentation-checkpoint", required=True)
    parser.add_argument("--attribute-checkpoint", required=True)
    parser.add_argument("--device", default="cuda")
    parser.add_argument("--attributes", nargs="+", default=None)
    parser.add_argument(
        "--top-k",
        type=int,
        default=None,
        help="Override the 3.1.3 checkpoint top-k output setting.",
    )
    parser.add_argument(
        "--confidence-threshold",
        type=float,
        default=None,
        help="Override the 3.1.3 checkpoint confidence threshold.",
    )
    parser.add_argument(
        "--output-dir",
        default=None,
        help=(
            "Write a delivery package with result.json, 3.1.1 visualization, "
            "3.1.2 mask, and 3.1.2 visualization. Explicit legacy output flags "
            "override their matching package paths."
        ),
    )
    parser.add_argument("--output", default=None)
    parser.add_argument("--mask-output", default=None)
    parser.add_argument("--vis-output", default=None)
    return parser.parse_args()


def resolve_artifact_paths(args: argparse.Namespace) -> dict[str, Path | None]:
    """Resolve legacy explicit paths and the optional delivery-package defaults."""
    output_dir = Path(args.output_dir) if args.output_dir else None

    def resolve(
        explicit: str | None,
        package_key: str,
    ) -> Path | None:
        if explicit:
            return Path(explicit)
        if output_dir is not None:
            return output_dir / PACKAGE_FILENAMES[package_key]
        return None

    return {
        "output_dir": output_dir,
        "result_json": resolve(args.output, "result_json"),
        "segmentation_visualization": (
            output_dir / PACKAGE_FILENAMES["segmentation_visualization"]
            if output_dir is not None
            else None
        ),
        "local_region_mask": resolve(args.mask_output, "local_region_mask"),
        "local_region_visualization": resolve(
            args.vis_output,
            "local_region_visualization",
        ),
    }


def artifact_payload(
    artifact_paths: dict[str, Path | None],
    *,
    has_local_region_mask: bool,
) -> dict[str, str | None]:
    """Serialize only artifacts that are expected to exist for this result."""
    payload = {key: _absolute_path(path) for key, path in artifact_paths.items()}
    if not has_local_region_mask:
        payload["local_region_mask"] = None
    return payload


def build_payload(
    *,
    image: str,
    query: str,
    result: Any,
    artifacts: dict[str, str | None],
) -> dict[str, Any]:
    """Build the versioned, JSON-serializable delivery result."""
    return {
        "schema": {
            "name": RESULT_SCHEMA_NAME,
            "version": RESULT_SCHEMA_VERSION,
        },
        "image": str(Path(image)),
        "query": query,
        "artifacts": artifacts,
        **result.to_dict(),
    }


def _absolute_path(path: Path | None) -> str | None:
    return str(path.resolve()) if path is not None else None


def write_delivery_artifacts(
    *,
    image: str | Path,
    query: str,
    result: Any,
    artifact_paths: dict[str, Path | None],
) -> dict[str, Any]:
    """Write one complete visual-pipeline delivery package."""
    local_region_mask = artifact_paths["local_region_mask"]
    has_local_region_mask = result.local_region.proposal is not None
    if local_region_mask is not None and has_local_region_mask:
        save_mask(result.local_region.proposal.proposal.mask, local_region_mask)

    segmentation_visualization = artifact_paths["segmentation_visualization"]
    if segmentation_visualization is not None:
        draw_segmentation_result(
            image,
            result.segmentation,
            segmentation_visualization,
        )

    local_region_visualization = artifact_paths["local_region_visualization"]
    if local_region_visualization is not None:
        draw_local_region_result(
            Path(image),
            result.local_region,
            local_region_visualization,
        )

    payload = build_payload(
        image=str(image),
        query=query,
        result=result,
        artifacts=artifact_payload(
            artifact_paths,
            has_local_region_mask=has_local_region_mask,
        ),
    )
    serialized = json.dumps(payload, ensure_ascii=False, indent=2)
    output_path = artifact_paths["result_json"]
    if output_path is not None:
        output_path.parent.mkdir(parents=True, exist_ok=True)
        output_path.write_text(serialized, encoding="utf-8")
    else:
        print(serialized)
    return payload


def main() -> None:
    args = parse_args()
    artifact_paths = resolve_artifact_paths(args)
    segmentation_config = load_config(args.segmentation_config)
    segmentation_predictor = FashionInstanceSegmentationPredictor(
        segmentation_config,
        checkpoint_path=args.segmentation_checkpoint,
        device=args.device,
    )
    attribute_predictor = FashionAttributePredictor(
        args.attribute_checkpoint,
        device=args.device,
        top_k=args.top_k,
        confidence_threshold=args.confidence_threshold,
    )
    pipeline = FashionVisualPipeline(segmentation_predictor, attribute_predictor)
    result = pipeline.predict(args.image, args.query, attributes=args.attributes)
    write_delivery_artifacts(
        image=args.image,
        query=args.query,
        result=result,
        artifact_paths=artifact_paths,
    )


if __name__ == "__main__":
    main()
